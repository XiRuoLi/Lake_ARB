# Step 1. Build DIAMOND database from the MobileOG protein set

diamond makedb \
    --in /path/to/mobileOG-db.faa \
    --db mobileOG-MC


# Step 2. Annotate MGE-associated proteins
# using DIAMOND and the MobileOG database

diamond blastp \
    --threads 64 \
    --query /path/to/proteins.faa \
    --db /path/to/mobileOG-MC.dmnd \
    --outfmt 6 qseqid sseqid pident length mismatch gapopen qstart qend sstart send evalue bitscore slen \
    --id 80 \
    --query-cover 70 \
    --evalue 1E-10 \
    -o mge_mobileOG.tsv
